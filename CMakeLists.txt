cmake_minimum_required(VERSION 3.16)
project(SoapyFifiSdr CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
find_package(SoapySDR CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBUSB REQUIRED IMPORTED_TARGET libusb-1.0)
pkg_check_modules(ALSA REQUIRED IMPORTED_TARGET alsa)
SOAPY_SDR_MODULE_UTIL(TARGET fifisdrSupport SOURCES SoapyFifiSdr.cpp LIBRARIES PkgConfig::LIBUSB PkgConfig::ALSA)

option(FIFISDR_BUILD_TESTS "Build the interpolator and DC blocker tests" ON)
if(FIFISDR_BUILD_TESTS)
    add_executable(fifi_interpolator_test fifi_interpolator_test.cpp)
    target_compile_options(fifi_interpolator_test PRIVATE -Wall -Wextra)
    add_executable(fifi_dc_blocker_test fifi_dc_blocker_test.cpp)
    target_compile_options(fifi_dc_blocker_test PRIVATE -Wall -Wextra)
endif()
